function doPost(e) {
  let payload = null;
  try {
    payload = JSON.parse(e && e.postData && e.postData.contents || '{}');
  } catch (_) {}

  // New website registration actions. The bridge verifies its own secret.
  if (payload && (payload.action === 'website_submit' ||
                  payload.action === 'website_status')) {
    return mcWebsiteBridgePost_(payload);
  }

  // Existing protected Donor Desk actions, unchanged.
  if (payload && (payload.action === 'desk_verify' ||
                  payload.action === 'desk_snapshot')) {
    return mcDeskBridgePost_(payload);
  }

  // Existing social-registration workflow, unchanged.
  return mcSocialBridgePost_(e);
}
